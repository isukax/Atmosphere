  "D:/Windows Kits/10/bin/10.0.26100.0/x64" "C:/Program Files (x86)/Windows Kits/10/bin/10.0.26100.0/x64" REQUIRED)
set(COMPACT_SHADER_DIR "${CMAKE_CURRENT_BINARY_DIR}/shaders")
file(MAKE_DIRECTORY "${COMPACT_SHADER_DIR}")
file(GLOB_RECURSE shader_headers CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/*.hlsli")
function(compact_shader source entry profile name)
  set(output "${COMPACT_SHADER_DIR}/${name}.cso")
  add_custom_command(OUTPUT "${output}" COMMAND "${DXC_EXECUTABLE}" -nologo -HV 2021 -WX -Zi -Qembed_debug
    -I "${CMAKE_CURRENT_SOURCE_DIR}/shaders" -T ${profile} -E ${entry} -Fo "${output}"
    "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${source}" DEPENDS "shaders/${source}" ${shader_headers} VERBATIM)
  set(shader_outputs ${shader_outputs} "${output}" PARENT_SCOPE)
endfunction()
foreach(pass TransmittanceLut MultiScatteringLut SkyViewLut AerialPerspective SkyIrradiance SkyFromLut)
  compact_shader(atmosphere/${pass}.hlsl CSMain cs_6_6 ${pass}_cs)
endforeach()

foreach(target atmosphere_compact_gfx atmosphere_compact_core)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /Zc:__cplusplus)
